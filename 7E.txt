import java.io.*;
import java.util.*;

public class Main {

    static class Macro {
        String expr;
        int state = -1;

        Macro(String expr) {
            this.expr = expr;
        }
    }

    static Map<String, Macro> macros = new HashMap<>();

    /*
        States:

        0 -> unsafe everywhere
        1 -> unsafe when:
             - used with * or /
             - appears on right side of -

        2 -> unsafe only when:
             - appears on right side of /

        3 -> completely safe
    */

    static int combine(int left, char op, int right) {

        // Any completely unsafe subexpression makes the whole
        // expression unsafe.
        if (left == 0 || right == 0)
            return 0;

        if (op == '+') {
            return 1;
        }

        if (op == '-') {
            // Right side of '-' must be completely safe
            // against +/-
            if (right == 1)
                return 0;

            return 1;
        }

        if (op == '*') {
            // Both sides must be safe from being split by
            // multiplication.
            if (left <= 1 || right <= 1)
                return 0;

            return 2;
        }

        // op == '/'
        if (left <= 1 || right <= 1)
            return 0;

        // a / (b * c) is not equivalent to a / b * c
        if (right == 2)
            return 0;

        return 2;
    }

    static boolean isOperator(char c) {
        return c == '+' || c == '-' || c == '*' || c == '/';
    }

    static String clean(String s) {
        return s.replaceAll("\\s+", "");
    }

    /*
     * Find whether the whole string is enclosed by one
     * pair of parentheses.
     */
    static boolean fullyParenthesized(String s) {

        if (s.length() < 2)
            return false;

        if (s.charAt(0) != '(' ||
            s.charAt(s.length() - 1) != ')')
            return false;

        int balance = 0;

        for (int i = 0; i < s.length(); i++) {

            if (s.charAt(i) == '(')
                balance++;

            else if (s.charAt(i) == ')')
                balance--;

            if (balance == 0 && i != s.length() - 1)
                return false;
        }

        return balance == 0;
    }

    static int solve(String original) {

        String s = clean(original);

        if (s.isEmpty())
            return 0;

        /*
         * First search for top-level + or -.
         *
         * We scan from right to left because these are the
         * lowest-precedence operators.
         */
        int balance = 0;

        for (int i = s.length() - 1; i >= 0; i--) {

            char c = s.charAt(i);

            if (c == ')') {
                balance++;
            }
            else if (c == '(') {
                balance--;
            }
            else if (balance == 0 &&
                    (c == '+' || c == '-')) {

                int left = solve(s.substring(0, i));
                int right = solve(s.substring(i + 1));

                return combine(left, c, right);
            }
        }

        /*
         * Now search for top-level * or /.
         */
        balance = 0;

        for (int i = s.length() - 1; i >= 0; i--) {

            char c = s.charAt(i);

            if (c == ')') {
                balance++;
            }
            else if (c == '(') {
                balance--;
            }
            else if (balance == 0 &&
                    (c == '*' || c == '/')) {

                int left = solve(s.substring(0, i));
                int right = solve(s.substring(i + 1));

                return combine(left, c, right);
            }
        }

        /*
         * If the entire expression is enclosed in parentheses,
         * its value becomes protected from the outside.

         * However, the inside still has to be valid.
         */
        if (fullyParenthesized(s)) {

            int inside = solve(
                    s.substring(1, s.length() - 1)
            );

            if (inside == 0)
                return 0;

            return 3;
        }

        /*
         * No operator and no surrounding parentheses:
         * variable / number / macro.
         */
        if (macros.containsKey(s)) {

            Macro m = macros.get(s);

            if (m.state == -1)
                m.state = solve(m.expr);

            return m.state;
        }

        // Ordinary variable or number.
        return 3;
    }

    static void readMacro(String line) {

        int p = line.indexOf("define");

        String rest = line.substring(p + 6).trim();

        int i = 0;

        while (i < rest.length() &&
                !Character.isWhitespace(rest.charAt(i))) {
            i++;
        }

        String name = rest.substring(0, i);

        String expression = rest.substring(i).trim();

        macros.put(name, new Macro(expression));
    }

    public static void main(String[] args) throws Exception {

        BufferedReader br =
                new BufferedReader(
                        new InputStreamReader(System.in)
                );

        int n = Integer.parseInt(br.readLine().trim());

        for (int i = 0; i < n; i++) {
            readMacro(br.readLine());
        }

        String expression = br.readLine();

        int result = solve(expression);

        System.out.println(
                result == 0 ? "Suspicious" : "OK"
        );
    }
}
