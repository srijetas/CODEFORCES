import java.io.*;
import java.util.*;

public class Main {

    static class Circle {
        double x, y, r;

        Circle(double x, double y, double r) {
            this.x = x;
            this.y = y;
            this.r = r;
        }
    }

    static final double EPS = 1e-10;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        Circle[] c = new Circle[3];

        for (int i = 0; i < 3; i++) {
            c[i] = new Circle(
                    sc.nextDouble(),
                    sc.nextDouble(),
                    sc.nextDouble()
            );
        }

        /*
         * We need:
         *
         *   dist(P, C0)^2 / r0^2
         * = dist(P, C1)^2 / r1^2
         * = dist(P, C2)^2 / r2^2
         * = k
         *
         * Therefore:
         *
         *   dist(P, Ci)^2 - dist(P, C0)^2
         *       = k * (ri^2 - r0^2)
         *
         * Expanding:
         *
         * 2(xi-x0)x + 2(yi-y0)y
         *   = xi^2 + yi^2 - x0^2 - y0^2
         *     - k(ri^2-r0^2)
         *
         * So x and y are linear functions of k:
         *
         *   x = ax*k + bx
         *   y = ay*k + by
         */

        double[][] A = new double[2][2];
        double[] B = new double[2];
        double[] K = new double[2];

        double s0 = c[0].x * c[0].x + c[0].y * c[0].y;

        for (int i = 1; i <= 2; i++) {
            double si = c[i].x * c[i].x + c[i].y * c[i].y;

            A[i - 1][0] = 2.0 * (c[i].x - c[0].x);
            A[i - 1][1] = 2.0 * (c[i].y - c[0].y);

            B[i - 1] = si - s0;

            K[i - 1] = -(c[i].r * c[i].r - c[0].r * c[0].r);
        }

        // A*x + ... = B + K*k
        //
        // Solve:
        // A00*x + A01*y = B0 + K0*k
        // A10*x + A11*y = B1 + K1*k

        double det =
                A[0][0] * A[1][1]
                - A[0][1] * A[1][0];

        if (Math.abs(det) < EPS) {
            return;
        }

        // x = px*k + qx
        double px =
                (K[0] * A[1][1] - A[0][1] * K[1])
                / det;

        double qx =
                (B[0] * A[1][1] - A[0][1] * B[1])
                / det;

        // y = py*k + qy
        double py =
                (A[0][0] * K[1] - K[0] * A[1][0])
                / det;

        double qy =
                (A[0][0] * B[1] - B[0] * A[1][0])
                / det;

        /*
         * Now:
         *
         * x = px*k + qx
         * y = py*k + qy
         *
         * And:
         *
         * (x-x0)^2 + (y-y0)^2 = k*r0^2
         *
         * This becomes:
         *
         * a*k^2 + b*k + c = 0
         */

        double ux = qx - c[0].x;
        double uy = qy - c[0].y;

        double qa = px * px + py * py;

        double qb =
                2.0 * (px * ux + py * uy)
                - c[0].r * c[0].r;

        double qc = ux * ux + uy * uy;

        double discriminant = qb * qb - 4.0 * qa * qc;

        if (discriminant < -EPS) {
            // No solution.
            return;
        }

        discriminant = Math.max(0.0, discriminant);

        double bestK = Double.POSITIVE_INFINITY;
        double answerX = 0;
        double answerY = 0;

        if (Math.abs(qa) < EPS) {
            // Linear equation.
            if (Math.abs(qb) < EPS) {
                return;
            }

            double k = -qc / qb;

            if (k > 1.0 + EPS) {
                answerX = px * k + qx;
                answerY = py * k + qy;
                bestK = k;
            }

        } else {
            double sqrtD = Math.sqrt(discriminant);

            double k1 = (-qb - sqrtD) / (2.0 * qa);
            double k2 = (-qb + sqrtD) / (2.0 * qa);

            double[] candidates = {k1, k2};

            for (double k : candidates) {

                /*
                 * k = dist^2 / r^2.
                 *
                 * The observer must be outside every stadium,
                 * so k > 1.
                 *
                 * The observation angle is:
                 *
                 *   2 * asin(1 / sqrt(k))
                 *
                 * Thus smaller k means larger angle.
                 */
                if (k <= 1.0 + EPS)
                    continue;

                if (k < bestK) {
                    bestK = k;
                    answerX = px * k + qx;
                    answerY = py * k + qy;
                }
            }
        }

        if (bestK == Double.POSITIVE_INFINITY) {
            // No valid observation point.
            return;
        }

        System.out.printf(
                Locale.US,
                "%.5f %.5f%n",
                answerX,
                answerY
        );
    }
}
