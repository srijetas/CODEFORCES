import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        String s = br.readLine();
        int n = s.length();

        int[] stack = new int[n + 1];
        int top = 0;

        // Base boundary
        stack[0] = -1;

        int maxLen = 0;
        int count = 0;

        for (int i = 0; i < n; i++) {

            if (s.charAt(i) == '(') {
                // Push index
                stack[++top] = i;
            } 
            else {
                // Pop matching '('
                top--;

                if (top < 0) {
                    // This ')' cannot be matched
                    top = 0;
                    stack[0] = i;
                } 
                else {
                    int len = i - stack[top];

                    if (len > maxLen) {
                        maxLen = len;
                        count = 1;
                    } 
                    else if (len == maxLen) {
                        count++;
                    }
                }
            }
        }

        if (maxLen == 0) {
            System.out.println("0 1");
        } else {
            System.out.println(maxLen + " " + count);
        }
    }
}
