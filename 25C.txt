import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();

        long[][] dist = new long[n][n];

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                dist[i][j] = sc.nextLong();
            }
        }

        int k = sc.nextInt();

        for (int x = 0; x < k; x++) {
            int a = sc.nextInt() - 1;
            int b = sc.nextInt() - 1;
            int c = sc.nextInt();

            long ans = 0;

            for (int i = 0; i < n; i++) {
                for (int j = i + 1; j < n; j++) {

                    long newDist = Math.min(
                        dist[i][j],
                        Math.min(
                            dist[i][a] + c + dist[b][j],
                            dist[i][b] + c + dist[a][j]
                        )
                    );

                    dist[i][j] = newDist;
                    dist[j][i] = newDist;
                }
            }

            for (int i = 0; i < n; i++) {
                for (int j = i + 1; j < n; j++) {
                    ans += dist[i][j];
                }
            }

            System.out.print(ans + " ");
        }

        sc.close();
    }
}
