import java.util.*;

public class Main {

    public static void main(String[] args) {

        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();
        int m = scanner.nextInt();

        int[][] boxes = new int[m][2];

        for (int row = 0; row < m; row++) {
            boxes[row][0] = scanner.nextInt(); // ai
            boxes[row][1] = scanner.nextInt(); // bi
        }

        // Sort by matches per box (bi) in descending order
        Arrays.sort(boxes, (first, second) -> {
            return Integer.compare(second[1], first[1]);
        });

        long answer = 0;

        for (int row = 0; row < m && n > 0; row++) {

            int availableBoxes = boxes[row][0];
            int matchesPerBox = boxes[row][1];

            int boxesTaken = Math.min(n, availableBoxes);

            answer += (long) boxesTaken * matchesPerBox;

            n -= boxesTaken;
        }

        System.out.println(answer);
    }
}
