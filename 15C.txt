import java.io.*;
import java.util.*;

public class Main {

    // XOR of all numbers from 1 to n
    static long xor1ToN(long n) {
        if (n < 0) {
            return 0;
        }

        switch ((int) (n % 4)) {
            case 0:
                return n;
            case 1:
                return 1;
            case 2:
                return n + 1;
            default:
                return 0;
        }
    }

    public static void main(String[] args) throws Exception {

        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

        int n = Integer.parseInt(br.readLine().trim());

        long totalXor = 0;

        for (int i = 0; i < n; i++) {

            StringTokenizer st = new StringTokenizer(br.readLine());

            long x = Long.parseLong(st.nextToken());
            long m = Long.parseLong(st.nextToken());

            // Last dumper contains:
            // x + m - 1 stones
            long last = x + m - 1;

            // XOR of x, x+1, ..., last
            long quarryXor =
                    xor1ToN(last) ^ xor1ToN(x - 1);

            totalXor ^= quarryXor;
        }

        if (totalXor != 0) {
            System.out.println("tolik");
        } else {
            System.out.println("bolik");
        }
    }
}
