import java.util.*;

public class Main {
    static final long INF = Long.MAX_VALUE / 4;

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int m = sc.nextInt();

        long[][] dist = new long[n][n];
        int[] degree = new int[n];
        long total = 0;

        for (int row = 0; row < n; row++) {
            Arrays.fill(dist[row], INF);
            dist[row][row] = 0;
        }

        for (int row = 0; row < m; row++) {
            int x = sc.nextInt() - 1;
            int y = sc.nextInt() - 1;
            long w = sc.nextLong();

            total += w;

            if (x != y) {
                degree[x]++;
                degree[y]++;
                dist[x][y] = Math.min(dist[x][y], w);
                dist[y][x] = Math.min(dist[y][x], w);
            } else {
                degree[x] += 2;
            }
        }

        if (m == 0) {
            System.out.println(0);
            return;
        }

        // Floyd-Warshall
        for (int mid = 0; mid < n; mid++) {
            for (int row = 0; row < n; row++) {
                for (int col = 0; col < n; col++) {
                    if (dist[row][mid] != INF &&
                        dist[mid][col] != INF) {
                        dist[row][col] = Math.min(
                            dist[row][col],
                            dist[row][mid] + dist[mid][col]
                        );
                    }
                }
            }
        }

        // Check connectivity of all vertices incident to an edge
        boolean[] active = new boolean[n];

        for (int row = 0; row < n; row++) {
            if (degree[row] > 0) {
                active[row] = true;
            }
        }

        for (int row = 0; row < n; row++) {
            if (active[row] && dist[0][row] == INF) {
                System.out.println(-1);
                return;
            }
        }

        // Collect odd-degree vertices
        ArrayList<Integer> odd = new ArrayList<>();

        for (int row = 0; row < n; row++) {
            if (degree[row] % 2 == 1) {
                odd.add(row);
            }
        }

        int k = odd.size();
        long[] dp = new long[1 << k];
        Arrays.fill(dp, INF);
        dp[0] = 0;

        for (int mask = 0; mask < (1 << k); mask++) {
            if (dp[mask] == INF) {
                continue;
            }

            int first = -1;

            for (int row = 0; row < k; row++) {
                if ((mask & (1 << row)) == 0) {
                    first = row;
                    break;
                }
            }

            if (first == -1) {
                continue;
            }

            for (int col = first + 1; col < k; col++) {
                if ((mask & (1 << col)) == 0) {
                    int nextMask = mask
                            | (1 << first)
                            | (1 << col);

                    dp[nextMask] = Math.min(
                        dp[nextMask],
                        dp[mask] + dist[odd.get(first)][odd.get(col)]
                    );
                }
            }
        }

        System.out.println(total + dp[(1 << k) - 1]);
    }
}
