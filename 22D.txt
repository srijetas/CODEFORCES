import java.util.*;

public class Main {
public static void main(String[] args) {
Scanner sc = new Scanner(System.in);

    int n = sc.nextInt();
    int[][] arr = new int[n][2];
    for (int row = 0; row < n; row++) {
        int a = sc.nextInt();
        int b = sc.nextInt();
        arr[row][0] = Math.min(a, b);
        arr[row][1] = Math.max(a, b);
    }
    Arrays.sort(arr, (a, b) -> a[1] - b[1]);
    ArrayList<Integer> nails = new ArrayList<>();
    int nail = Integer.MIN_VALUE;
    for (int row = 0; row < n; row++) {
        if (nail < arr[row][0] || nail > arr[row][1]) {
            nail = arr[row][1];
            nails.add(nail);
        }
    }
    System.out.println(nails.size());
    for (int x : nails) {
        System.out.print(x + " ");
    }
}

}
