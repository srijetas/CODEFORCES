import java.io.*;

public class Main {

    static class FastScanner {
        private final InputStream in = System.in;
        private final byte[] buffer = new byte[1 << 16];
        private int ptr = 0;
        private int len = 0;

        private int read() throws IOException {
            if (ptr >= len) {
                len = in.read(buffer);
                ptr = 0;
                if (len <= 0) return -1;
            }
            return buffer[ptr++];
        }

        int nextInt() throws IOException {
            int c;

            do {
                c = read();
            } while (c <= ' ');

            int num = 0;

            while (c > ' ') {
                num = num * 10 + (c - '0');
                c = read();
            }

            return num;
        }
    }

    public static void main(String[] args) throws Exception {

        FastScanner fs = new FastScanner();

        int N = fs.nextInt();
        int K = fs.nextInt();

        boolean[][] occupied = new boolean[K][K];

        // Center using 0-based indexing
        int center = K / 2;

        StringBuilder answer = new StringBuilder();

        for (int query = 0; query < N; query++) {

            int M = fs.nextInt();

            int bestRow = -1;
            int bestLeft = -1;
            int bestCost = Integer.MAX_VALUE;

            for (int row = 0; row < K; row++) {

                for (int left = 0; left + M <= K; left++) {

                    boolean possible = true;
                    int cost = 0;

                    for (int seat = left; seat < left + M; seat++) {

                        if (occupied[row][seat]) {
                            possible = false;
                            break;
                        }

                        // Manhattan distance from center
                        cost += Math.abs(row - center)
                              + Math.abs(seat - center);
                    }

                    if (!possible) continue;

                    /*
                     * Loops go:
                     * smaller row first
                     * then smaller left first
                     *
                     * Therefore, only strict improvement is needed.
                     */
                    if (cost < bestCost) {
                        bestCost = cost;
                        bestRow = row;
                        bestLeft = left;
                    }
                }
            }

            if (bestRow == -1) {
                answer.append("-1\n");
            } else {

                for (int seat = bestLeft; seat < bestLeft + M; seat++) {
                    occupied[bestRow][seat] = true;
                }

                answer.append(bestRow + 1)
                      .append(" ")
                      .append(bestLeft + 1)
                      .append(" ")
                      .append(bestLeft + M)
                      .append("\n");
            }
        }

        System.out.print(answer);
    }
}
