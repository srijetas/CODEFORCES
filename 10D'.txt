import java.util.*;
import java.io.*;
import java.util.StringTokenizer;

public class Main {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        
        int n = Integer.parseInt(br.readLine().trim());
        long[] a = new long[n + 1];
        StringTokenizer st = new StringTokenizer(br.readLine());
        for (int i = 1; i <= n; i++) a[i] = Long.parseLong(st.nextToken());
        
        int m = Integer.parseInt(br.readLine().trim());
        long[] b = new long[m + 1];
        st = new StringTokenizer(br.readLine());
        for (int i = 1; i <= m; i++) b[i] = Long.parseLong(st.nextToken());
        
        int[] dp = new int[m + 1];
        int[] parent = new int[m + 1];
        Arrays.fill(parent, -1);
        
        for (int i = 1; i <= n; i++) {
            int best = 0;
            int bestJ = -1;
            for (int j = 1; j <= m; j++) {
                if (a[i] == b[j]) {
                    if (best + 1 > dp[j]) {
                        dp[j] = best + 1;
                        parent[j] = bestJ;
                    }
                }
                if (b[j] < a[i]) {
                    if (dp[j] > best) {
                        best = dp[j];
                        bestJ = j;
                    }
                }
            }
        }
        
        int pos = -1, mx = 0;
        for (int j = 1; j <= m; j++) {
            if (dp[j] > mx) {
                mx = dp[j];
                pos = j;
            }
        }
        
        ArrayList<Long> res = new ArrayList<>();
        while (pos != -1) {
            res.add(b[pos]);
            pos = parent[pos];
        }
        Collections.reverse(res);
        
        StringBuilder sb = new StringBuilder();
        sb.append(mx).append("\n");
        for (int i = 0; i < res.size(); i++) {
            sb.append(res.get(i));
            if (i + 1 < res.size()) sb.append(" ");
        }
        sb.append("\n");
        
        System.out.print(sb);
    }
}
