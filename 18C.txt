import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int[] a = new int[n];

        long total = 0;

        for (int i = 0; i < n; i++) {
            a[i] = sc.nextInt();
            total += a[i];
        }

        if (total % 2 != 0) {
            System.out.println(0);
            return;
        }

        long prefix = 0;
        int count = 0;

        for (int i = 0; i < n - 1; i++) {
            prefix += a[i];

            if (prefix == total / 2) {
                count++;
            }
        }

        System.out.println(count);
    }
}
