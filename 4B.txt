
import java.util.*;

public class Main {
    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        int d = sc.nextInt();
        int sumTime = sc.nextInt();

        int[] minTime = new int[d];
        int[] maxTime = new int[d];

        int minSum = 0;
        int maxSum = 0;

        for (int i = 0; i < d; i++) {
            minTime[i] = sc.nextInt();
            maxTime[i] = sc.nextInt();

            minSum += minTime[i];
            maxSum += maxTime[i];
        }

        // Impossible
        if (sumTime < minSum || sumTime > maxSum) {
            System.out.println("NO");
            return;
        }

        int[] schedule = new int[d];

        // Initially give minimum hours
        for (int i = 0; i < d; i++) {
            schedule[i] = minTime[i];
        }

        // Extra hours needed
        int remaining = sumTime - minSum;

        // Distribute extra hours
        for (int i = 0; i < d; i++) {

            int canAdd = maxTime[i] - minTime[i];

            int add = Math.min(remaining, canAdd);

            schedule[i] += add;
            remaining -= add;
        }

        System.out.println("YES");

        for (int i = 0; i < d; i++) {
            System.out.print(schedule[i] + " ");
        }
    }
}
