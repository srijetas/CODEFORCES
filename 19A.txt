import java.util.*;

public class Main {
static class Team {
String name;
int points = 0;
int scored = 0;
int missed = 0;

    Team(String name) {
        this.name = name;
    }
    int goalDifference() {
        return scored - missed;
    }
}
public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = Integer.parseInt(sc.nextLine());
    Team[] teams = new Team[n];
    Map<String, Team> map = new HashMap<>();
    for (int i = 0; i < n; i++) {
        String name = sc.nextLine();
        teams[i] = new Team(name);
        map.put(name, teams[i]);
    }
    int matches = n * (n - 1) / 2;
    for (int i = 0; i < matches; i++) {
        String line = sc.nextLine();
        String[] parts = line.split(" ");
        String[] names = parts[0].split("-");
        String[] goals = parts[1].split(":");
        Team team1 = map.get(names[0]);
        Team team2 = map.get(names[1]);
        int goal1 = Integer.parseInt(goals[0]);
        int goal2 = Integer.parseInt(goals[1]);
        team1.scored += goal1;
        team1.missed += goal2;
        team2.scored += goal2;
        team2.missed += goal1;
        if (goal1 > goal2) {
            team1.points += 3;
        } else if (goal1 < goal2) {
            team2.points += 3;
        } else {
            team1.points++;
            team2.points++;
        }
    }
    Arrays.sort(teams, (a, b) -> {
        if (a.points != b.points) {
            return b.points - a.points;
        }
        if (a.goalDifference() != b.goalDifference()) {
            return b.goalDifference() - a.goalDifference();
        }
        return b.scored - a.scored;
    });
    String[] qualified = new String[n / 2];
    for (int i = 0; i < n / 2; i++) {
        qualified[i] = teams[i].name;
    }
    Arrays.sort(qualified);
    for (String name : qualified) {
        System.out.println(name);
    }
    sc.close();
}

}
