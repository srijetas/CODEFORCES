import java.io.*;
import java.util.*;

public class Main {

    static class Node {
        long diff;
        int index;

        Node(long diff, int index) {
            this.diff = diff;
            this.index = index;
        }
    }

    public static void main(String[] args) throws Exception {

        BufferedReader br =
                new BufferedReader(new InputStreamReader(System.in));

        String s = br.readLine();
        int n = s.length();

        char[] ans = s.toCharArray();

        long cost = 0;
        int balance = 0;

        PriorityQueue<Node> pq = new PriorityQueue<>(
                (a, b) -> Long.compare(a.diff, b.diff)
        );

        int qCount = 0;

        for (int i = 0; i < n; i++) {

            char ch = ans[i];

            if (ch == '(') {
                balance++;
            }

            else if (ch == ')') {
                balance--;
            }

            else {
                // Read cost of this ?
                StringTokenizer st =
                        new StringTokenizer(br.readLine());

                long a = Long.parseLong(st.nextToken());
                long b = Long.parseLong(st.nextToken());

                // Initially choose ')'
                ans[i] = ')';

                cost += b;
                balance--;

                // Difference if we change ')' -> '('
                long diff = a - b;

                // IMPORTANT:
                // Add only after processing this position.
                pq.add(new Node(diff, i));

                qCount++;
            }

            // If balance becomes negative,
            // change the cheapest previous '?' to '('.
            if (balance < 0) {

                if (pq.isEmpty()) {
                    System.out.println(-1);
                    return;
                }

                Node best = pq.poll();

                ans[best.index] = '(';

                cost += best.diff;

                // ')' -> '(' increases balance by 2
                balance += 2;
            }
        }

        // A regular bracket sequence must end with balance 0
        if (balance != 0) {
            System.out.println(-1);
            return;
        }

        System.out.println(cost);
        System.out.println(new String(ans));
    }
}
