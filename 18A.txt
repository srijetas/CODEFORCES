import java.util.Scanner;

public class Main {

    static boolean isRight(int[] x, int[] y) {
        long a = dist(x[0], y[0], x[1], y[1]);
        long b = dist(x[1], y[1], x[2], y[2]);
        long c = dist(x[2], y[2], x[0], y[0]);

        if (a == 0 || b == 0 || c == 0) {
            return false;
        }

        return a + b == c || b + c == a || c + a == b;
    }

    static long dist(int x1, int y1, int x2, int y2) {
        long dx = x1 - x2;
        long dy = y1 - y2;
        return dx * dx + dy * dy;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int[] x = new int[3];
        int[] y = new int[3];

        for (int i = 0; i < 3; i++) {
            x[i] = sc.nextInt();
            y[i] = sc.nextInt();
        }

        if (isRight(x, y)) {
            System.out.println("RIGHT");
            return;
        }

        int[][] moves = {
            {1, 0}, {-1, 0}, {0, 1}, {0, -1}
        };

        for (int i = 0; i < 3; i++) {
            int oldX = x[i];
            int oldY = y[i];

            for (int[] move : moves) {
                x[i] = oldX + move[0];
                y[i] = oldY + move[1];

                if (isRight(x, y)) {
                    System.out.println("ALMOST");
                    return;
                }
            }

            x[i] = oldX;
            y[i] = oldY;
        }

        System.out.println("NEITHER");
    }
}
