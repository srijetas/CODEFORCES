import java.util.*;

public class Main {

    static int n;
    static long k;
    static long[][][] dp = new long[55][2][2];
    static int[] res = new int[55];

    static long solve(int pos, int rev, int inv) {
        if (dp[pos][rev][inv] != -1)
            return dp[pos][rev][inv];

        if (pos * 2 >= n)
            return dp[pos][rev][inv] = 1;

        long ans = 0;

        int left = pos + 1;
        int right = n - pos;

        for (int a = 0; a < 2; a++) {
            if (res[left] != -1 && res[left] != a)
                continue;

            for (int b = 0; b < 2; b++) {
                if (res[right] != -1 && res[right] != b)
                    continue;

                if (left == right && a != b)
                    continue;

                if (rev == 1 && a > b)
                    continue;

                if (inv == 1 && a > (1 - b))
                    continue;

                ans += solve(
                        pos + 1,
                        (rev == 1 && a == b) ? 1 : 0,
                        (inv == 1 && a == (1 - b)) ? 1 : 0
                );

                if (ans > 1000000000000000000L)
                    ans = 1000000000000000000L;
            }
        }

        return dp[pos][rev][inv] = ans;
    }

    static void resetDP() {
        for (int i = 0; i < 55; i++)
            for (int j = 0; j < 2; j++)
                Arrays.fill(dp[i][j], -1);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        n = sc.nextInt();
        k = sc.nextLong();

        k++; // skip the representative 000...0

        Arrays.fill(res, -1);

        resetDP();

        if (solve(0, 1, 1) < k) {
            System.out.println(-1);
            return;
        }

        StringBuilder answer = new StringBuilder();

        for (int i = 1; i <= n; i++) {
            res[i] = 0;

            resetDP();
            long cnt = solve(0, 1, 1);

            if (cnt < k) {
                k -= cnt;
                res[i] = 1;
            }

            answer.append(res[i]);
        }

        System.out.println(answer);
    }
}
