import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int m = sc.nextInt();
        char c = sc.next().charAt(0);

        char[][] grid = new char[n][m];

        for (int i = 0; i < n; i++) {
            String s = sc.next();
            grid[i] = s.toCharArray();
        }

        Set<Character> deputies = new HashSet<>();

        int[] dx = {-1, 1, 0, 0};
        int[] dy = {0, 0, -1, 1};

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {

                if (grid[i][j] != c)
                    continue;

                for (int k = 0; k < 4; k++) {
                    int ni = i + dx[k];
                    int nj = j + dy[k];

                    if (ni >= 0 && ni < n && nj >= 0 && nj < m) {
                        char x = grid[ni][nj];

                        if (x != '.' && x != c) {
                            deputies.add(x);
                        }
                    }
                }
            }
        }

        System.out.println(deputies.size());
    }
}
