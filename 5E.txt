import java.io.*;
import java.util.*;

public class Main {

    static class FastScanner {
        private final InputStream in = System.in;
        private final byte[] buffer = new byte[1 << 16];
        private int ptr = 0, len = 0;

        private int read() throws IOException {
            if (ptr >= len) {
                len = in.read(buffer);
                ptr = 0;
                if (len <= 0) return -1;
            }
            return buffer[ptr++];
        }

        int nextInt() throws IOException {
            int c;
            do {
                c = read();
            } while (c <= ' ');

            int res = 0;

            while (c > ' ') {
                res = res * 10 + (c - '0');
                c = read();
            }

            return res;
        }
    }

    public static void main(String[] args) throws Exception {

        FastScanner fs = new FastScanner();

        int n = fs.nextInt();

        int[] a = new int[n];

        // Find one maximum
        int maxIndex = 0;

        for (int i = 0; i < n; i++) {
            a[i] = fs.nextInt();

            if (a[i] > a[maxIndex]) {
                maxIndex = i;
            }
        }

        /*
         * Rotate array so that a maximum is first.
         */
        int[] h = new int[n];

        for (int i = 0; i < n; i++) {
            h[i] = a[(maxIndex + i) % n];
        }

        /*
         * stackHeight[top]
         * stackCount[top]
         *
         * Heights are decreasing.
         */
        int[] stackHeight = new int[n];
        int[] stackCount = new int[n];

        int top = 0;

        stackHeight[0] = h[0];
        stackCount[0] = 1;

        long ans = 0;

        for (int i = 1; i < n; i++) {

            int cur = h[i];

            /*
             * Pop all smaller heights.
             */
            while (top >= 0 && cur > stackHeight[top]) {
                ans += stackCount[top];
                top--;
            }

            /*
             * Stack became empty.
             */
            if (top < 0) {
                stackHeight[++top] = cur;
                stackCount[top] = 1;
                continue;
            }

            /*
             * Equal heights.
             */
            if (cur == stackHeight[top]) {

                ans += stackCount[top];

                stackCount[top]++;

                /*
                 * There is a higher hill on the left
                 * which can also see this group.
                 */
                if (top > 0) {
                    ans++;
                }
            }

            /*
             * Current hill is smaller than stack top.
             */
            else {
                ans++;

                stackHeight[++top] = cur;
                stackCount[top] = 1;
            }
        }

        /*
         * Handle the circular connection back to the
         * first (maximum) hill.
         */
        while (top > 1) {
            ans += stackCount[top];
            top--;
        }

        /*
         * Two remaining stack levels.
         *
         * If the lower level contains multiple equal
         * maximum-height hills, they can see the last group.
         */
        if (top == 1) {
            if (stackCount[0] > 1) {
                ans += stackCount[1];
            }
        }

        System.out.println(ans);
    }
}
