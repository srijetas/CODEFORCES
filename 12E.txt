import java.io.*;

public class Main {

    public static void main(String[] args) throws Exception {

        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        int n = Integer.parseInt(br.readLine().trim());

        int m = n - 1;              // odd
        int inv2 = (m + 1) / 2;     // inverse of 2 modulo m

        StringBuilder out = new StringBuilder(n * n * 2);

        for (int i = 0; i < n; i++) {

            for (int j = 0; j < n; j++) {

                int value;

                if (i == j) {
                    value = 0;
                }
                else if (i == m) {
                    // Last vertex connected to i
                    value = j + 1;
                }
                else if (j == m) {
                    // Last vertex connected to j
                    value = i + 1;
                }
                else {
                    // Edge coloring
                    value = ((i + j) * inv2) % m + 1;
                }

                if (j > 0) {
                    out.append(' ');
                }

                out.append(value);
            }

            out.append('\n');
        }

        System.out.print(out);
    }
}
