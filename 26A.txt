
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int ans = 0;

        for (int i = 2; i <= n; i++) {
            int num = i;
            int count = 0;

            for (int j = 2; j * j <= num; j++) {
                if (num % j == 0) {
                    count++;
                    while (num % j == 0) {
                        num /= j;
                    }
                }
            }

            if (num > 1) {
                count++;
            }

            if (count == 2) {
                ans++;
            }
        }

        System.out.println(ans);
        sc.close();
    }
}
