import java.io.*;
import java.util.*;

public class Main {

    static int n, m;
    static char[][] grid;

    // 8 directions
    static final int[] dx = {-1, -1, -1, 0, 0, 1, 1, 1};
    static final int[] dy = {-1, 0, 1, -1, 1, -1, 0, 1};

    // Count consecutive 1s in a direction, including (r,c)
    static int ray(int r, int c, int dr, int dc) {
        int len = 0;

        while (r >= 0 && r < n && c >= 0 && c < m
                && grid[r][c] == '1') {
            len++;
            r += dr;
            c += dc;
        }

        return len;
    }

    // Check square whose sides are parallel to matrix sides.
    static int normalSquare(int r, int c) {

        // Starting from top-left:
        // down -> bottom-left
        int a = ray(r, c, 1, 0);

        // bottom-left -> bottom-right
        int b = ray(r + a - 1, c, 0, 1);

        // bottom-right -> top-right
        int d = ray(r + a - 1, c + b - 1, -1, 0);

        // top-right -> top-left
        int e = ray(r, c + b - 1, 0, -1);

        if (a == b && b == d && d == e) {
            return a;
        }

        return 0;
    }

    // Check square whose sides are parallel to diagonals.
    static int diagonalSquare(int r, int c) {

        // Starting from top vertex:
        // down-left -> left vertex
        int a = ray(r, c, 1, -1);

        // left vertex -> bottom vertex
        int b = ray(r + a - 1, c - (a - 1), 1, 1);

        // bottom vertex -> right vertex
        int d = ray(r + 2 * (a - 1), c, -1, 1);

        // right vertex -> top vertex
        int e = ray(r + (a - 1), c + (a - 1), -1, -1);

        if (a == b && b == d && d == e) {
            return a;
        }

        return 0;
    }

    // Returns size of the 8-connected component and removes it.
    static int componentSize(int sr, int sc) {
        int total = n * m;

        int[] stackR = new int[total];
        int[] stackC = new int[total];

        int top = 0;
        stackR[top] = sr;
        stackC[top] = sc;
        top++;

        grid[sr][sc] = '0';

        int size = 0;

        while (top > 0) {
            top--;

            int r = stackR[top];
            int c = stackC[top];

            size++;

            for (int k = 0; k < 8; k++) {
                int nr = r + dx[k];
                int nc = c + dy[k];

                if (nr >= 0 && nr < n &&
                    nc >= 0 && nc < m &&
                    grid[nr][nc] == '1') {

                    grid[nr][nc] = '0';

                    stackR[top] = nr;
                    stackC[top] = nc;
                    top++;
                }
            }
        }

        return size;
    }

    public static void main(String[] args) throws Exception {
        FastScanner fs = new FastScanner(System.in);

        int t = fs.nextInt();
        StringBuilder out = new StringBuilder();

        while (t-- > 0) {
            n = fs.nextInt();
            m = fs.nextInt();

            grid = new char[n][m];

            for (int i = 0; i < n; i++) {
                grid[i] = fs.next().toCharArray();
            }

            int answer = 0;

            for (int i = 0; i < n; i++) {
                for (int j = 0; j < m; j++) {

                    if (grid[i][j] != '1')
                        continue;

                    // Since we scan row by row, the first cell
                    // of a square is its top-left/top vertex.
                    int normal = normalSquare(i, j);
                    int diagonal = diagonalSquare(i, j);

                    // Remove the whole 8-connected component.
                    int component = componentSize(i, j);

                    if (normal > 1 &&
                        component == 4 * (normal - 1)) {
                        answer++;
                    }
                    else if (diagonal > 1 &&
                             component == 4 * (diagonal - 1)) {
                        answer++;
                    }
                }
            }

            out.append(answer).append('\n');
        }

        System.out.print(out);
    }

    // Fast input
    static class FastScanner {
        private final InputStream in;
        private final byte[] buffer = new byte[1 << 16];
        private int ptr = 0, len = 0;

        FastScanner(InputStream in) {
            this.in = in;
        }

        private int read() throws IOException {
            if (ptr >= len) {
                len = in.read(buffer);
                ptr = 0;

                if (len <= 0)
                    return -1;
            }

            return buffer[ptr++];
        }

        String next() throws IOException {
            StringBuilder sb = new StringBuilder();
            int ch;

            do {
                ch = read();
            } while (ch <= ' ');

            while (ch > ' ') {
                sb.append((char) ch);
                ch = read();
            }

            return sb.toString();
        }

        int nextInt() throws IOException {
            return Integer.parseInt(next());
        }
    }
}
